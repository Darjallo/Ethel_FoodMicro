Build agent images from this directory:

sudo docker build -f agent_pool/agents/test_agent/Dockerfile -t test_agent:latest .

or

sudo docker build -f agent_pool/agents/emb_file_ada3large/Dockerfile -t emb_file_ada3large:latest .

or

sudo docker build -f agent_pool/agents/emb_similarity_ada3large/Dockerfile -t emb_similarity_ada3large:latest .

sudo docker build -t emb_ada3large:latest -f agent_pool/agents/emb_ada3large/Dockerfile .

sudo docker build -t file_to_text:latest -f agent_pool/agents/file_to_text/Dockerfile .

sudo docker build -t chunk_text:latest -f agent_pool/agents/chunk_text/Dockerfile .

sudo docker build -t store_vectors:latest -f agent_pool/agents/store_vectors/Dockerfile .

respectively.
